def build_models(cfg,seed):
    frozen_numerics=numerical_environment()
    backend=cfg.get('vllm_attention_backend','FLASH_ATTN')
    assert backend in {'FLASH_ATTN','XFORMERS'},'Unsupported inference attention backend'
    os.environ['VLLM_ATTENTION_BACKEND']=backend
    # Triton builds its Python extension on first use. Minimal host images may
    # omit Python.h; matching distro headers can be extracted without root.
    local_include=ROOT/'artifacts/build-deps/usr/include'
    if (local_include/'python3.12/Python.h').exists():
        include_paths=[str(local_include/'python3.12'),str(local_include)]
        if os.environ.get('C_INCLUDE_PATH'):include_paths.append(os.environ['C_INCLUDE_PATH'])
        os.environ['C_INCLUDE_PATH']=os.pathsep.join(include_paths)
    from vllm import LLM
    torch.set_num_threads(8)
    torch.manual_seed(seed)
    torch.cuda.manual_seed_all(seed)
    random.seed(seed)
    model_path=str(ROOT/cfg['model'])
    tokenizer=AutoTokenizer.from_pretrained(model_path,local_files_only=True)
    tokenizer.pad_token=tokenizer.eos_token
    actor=AutoModelForCausalLM.from_pretrained(model_path,torch_dtype=torch.float32,
                   attn_implementation='sdpa',local_files_only=True).to('cuda')
    actor.gradient_checkpointing_enable(gradient_checkpointing_kwargs={'use_reentrant':False})
    actor.config.use_cache=False
    actor._project_forward_bf16=cfg.get('project_forward_bf16',False)
    if actor._project_forward_bf16:
        actor.model._gradient_checkpointing_func=projected_checkpoint
    ref=AutoModelForCausalLM.from_pretrained(model_path,torch_dtype=torch.bfloat16,
                   attn_implementation='sdpa',local_files_only=True).to('cuda').eval()
    ref.requires_grad_(False)
    engine=LLM(model=model_path,tokenizer=model_path,dtype='bfloat16',
               tensor_parallel_size=1,distributed_executor_backend='uni',
               gpu_memory_utilization=cfg['vllm_memory_fraction'],
               max_model_len=cfg['max_context_tokens'],max_num_seqs=cfg['vllm_max_sequences'],
               max_num_batched_tokens=4096,enable_chunked_prefill=True,
               enable_prefix_caching=True,disable_async_output_proc=True,
               seed=seed,enforce_eager=cfg.get('vllm_enforce_eager',False),disable_log_stats=True)
    if numerical_environment()!=frozen_numerics:
        raise RuntimeError('Model/backend initialization changed frozen numerical controls')
    return actor,ref,engine,tokenizer


def inference_cache_boundary(engine, phase, *, reset=False):
    """Fail closed at actual vLLM V0/0.9.2 idle boundaries before cache changes."""
    backend=engine.llm_engine
    state=dict(phase=phase,unfinished_requests=backend.get_num_unfinished_requests(),
               has_unfinished_requests=backend.has_unfinished_requests(),schedulers=[])
    for scheduler in backend.scheduler:
        manager=scheduler.block_manager
        queues={name:len(getattr(scheduler,name)) for name in
                ('waiting','running','swapped','_async_stopped')}
        queues.update(block_tables=len(manager.block_tables),
                      cross_block_tables=len(manager.cross_block_tables))
        used={str(device):allocator.get_num_total_blocks()-allocator.get_num_free_blocks()
              for device,allocator in manager.block_allocator._allocators.items()}
        state['schedulers'].append(dict(queues=queues,used_blocks=used))
    busy=state['unfinished_requests'] or state['has_unfinished_requests'] or any(
        any(item['queues'].values()) or any(item['used_blocks'].values()) for item in state['schedulers'])
    if busy:
        raise RuntimeError('Inference boundary is not idle: '+json.dumps(state,sort_keys=True))
    state['prefix_cache_reset']=backend.reset_prefix_cache() if reset else None
    if reset and state['prefix_cache_reset'] is not True:
        raise RuntimeError('Prefix-cache reset failed: '+json.dumps(state,sort_keys=True))
    if reset:
        print(json.dumps({'INFERENCE_CACHE_BOUNDARY':state}),flush=True)
    return state


def sync_weights(actor,engine):
    # V0/UniProc colocates the inference model. No network service or extra GPU.
    inference_cache_boundary(engine,'before_weight_sync')
    vm=engine.llm_engine.model_executor.driver_worker.model_runner.model
    with torch.no_grad():
        loaded=vm.load_weights((name,p.detach().to(torch.bfloat16)) for name,p in actor.named_parameters())
        verify_qwen_weights(actor,engine)
    inference_cache_boundary(engine,'after_weight_sync',reset=True)
    torch.cuda.synchronize()
    return len(loaded) if loaded is not None else None


def selected_hidden(model,ids,attention,mask):
    with torch.autocast('cuda',dtype=torch.bfloat16):
        if getattr(model,'_project_forward_bf16',False):
            params={name:p.to(torch.bfloat16) for name,p in model.model.named_parameters()}
            hidden=functional_call(model.model,params,(),dict(input_ids=ids,attention_mask=attention,use_cache=False)).last_hidden_state
        else:
            hidden=model.model(input_ids=ids,attention_mask=attention,use_cache=False).last_hidden_state
        keep=mask[:,1:].bool()
        return hidden[:,:-1][keep],ids[:,1:][keep]


def configure_vocabulary(actor,ref,tokenizer,cfg,method):
    cfg['active_method']=method
    allowed=sampling_allowed_token_ids(tokenizer,cfg,method)
    actor._allowed_token_ids=allowed
    ref._allowed_token_ids=allowed
    return sampling_vocabulary_metadata(tokenizer,cfg,method)


def policy_logits(model,hidden):
    logits=model.lm_head(hidden).float()
    allowed=getattr(model,'_allowed_token_ids',None)
    if allowed is not None:
        keep=torch.zeros(logits.shape[-1],device=logits.device,dtype=torch.bool)
        keep[allowed]=True
        # Finite floor gives exactly zero probability and avoids -inf - -inf in KL.
        logits=logits.masked_fill(~keep,torch.finfo(logits.dtype).min)
    return logits


def selected_logprobs(model,ids,attention,mask,gradient=False):
    """Only allocate vocabulary logits for assistant targets, in bounded chunks."""
    with torch.autocast('cuda',dtype=torch.bfloat16):
        h,labels=selected_hidden(model,ids,attention,mask)
        parts=[]
        def head_logprob(x,y):
            with torch.autocast('cuda',dtype=torch.bfloat16):
                logits=policy_logits(model,x)
            return -F.cross_entropy(logits,y,reduction='none')
        for lo in range(0,len(labels),512):
            hs,ys=h[lo:lo+512],labels[lo:lo+512]
            lp=checkpoint(head_logprob,hs,ys,use_reentrant=False) if gradient else head_logprob(hs,ys)
            parts.append(lp)
        return torch.cat(parts)


def selected_exact_kl(actor,ref,ids,attention,mask):
    """Exact categorical KL at sampled contexts; no exp(ref_sample-current_sample) tail."""
    h,labels=selected_hidden(actor,ids,attention,mask)
    with torch.no_grad():rh,_=selected_hidden(ref,ids,attention,mask)
    def terms(x,y,z):
        with torch.autocast('cuda',dtype=torch.bfloat16):
            logits=policy_logits(actor,x)
            with torch.no_grad():ref_logits=policy_logits(ref,z)
        logp=F.log_softmax(logits,-1)
        logq=F.log_softmax(ref_logits,-1)
        lp=logp.gather(-1,y[:,None]).squeeze(-1)
        kl=(logp.exp()*(logp-logq)).sum(-1)
        return lp,kl
    lps=[];kls=[]
    for lo in range(0,len(labels),256):
        lp,kl=checkpoint(terms,h[lo:lo+256],labels[lo:lo+256],rh[lo:lo+256],use_reentrant=False)
        lps.append(lp);kls.append(kl)
    return torch.cat(lps),torch.cat(kls)


def pack(rows,pad):
    maxlen=max(len(b.ids) for b,_,_ in rows)
    ids=torch.full((len(rows),maxlen),pad,device='cuda',dtype=torch.long)
    attn=torch.zeros_like(ids)
    mask=torch.zeros_like(ids)
    advantages=[];weights=[];old=[]
    for i,(b,a,w) in enumerate(rows):
        n=len(b.ids)
        ids[i,:n]=torch.tensor(b.ids,device='cuda')
        attn[i,:n]=1
        mask[i,:n]=torch.tensor(b.mask,device='cuda')
        advantages.extend([a]*b.ntokens)
        weights.extend([w]*b.ntokens)
        old.extend(getattr(b,'training_old_logprobs',None) or [v for v,m in zip(b.old_logprobs,b.mask) if m])
    f=lambda x:torch.tensor(x,dtype=torch.float32,device='cuda')
    return ids,attn,mask,f(advantages),f(weights),f(old)


def microbatches(rows,cfg):
    # Sort only within the current optimizer minibatch to reduce padding.
    rows=sorted(rows,key=lambda x:len(x[0].ids))
    batch=[]
    for row in rows:
        if batch and (len(batch)>=cfg['micro_batch_size'] or
                      len(row[0].ids)*(len(batch)+1)>cfg['micro_batch_token_limit']):
            yield batch
            batch=[]
        batch.append(row)
    if batch:
        yield batch


def update(actor,ref,optimizer,branches,cfg,method,tokenizer,rng,n_tasks):
    actor.train()
    metrics=Counter();mb_count=0
    task_order=rng.permutation(n_tasks).tolist()
    stages=[]
    for off in range(0,n_tasks,cfg['mini_batch_tasks']):
        chosen=task_order[off:off+cfg['mini_batch_tasks']]
        idmap={old:new for new,old in enumerate(chosen)}
        bs=[]
        for b in branches:
            if b.task_id in idmap:
                c=copy.copy(b);c.task_id=idmap[b.task_id];bs.append(c)
        rows=training_weights(bs,len(chosen),method=='dapo',cfg=cfg)
        stages.append(list(microbatches(rows,cfg)))
    # Recompute the frozen training-policy anchor BEFORE either optimizer update.
    # Sampling logprobs remain unchanged in the raw record.
    mismatches=[]
    if cfg.get('recompute_old_logprobs',False):
        actor.eval()
        for chunks in stages:
            for chunk in chunks:
                ids,attn,mask,_,_,sampled=pack(chunk,tokenizer.pad_token_id)
                with torch.no_grad():anchor=selected_logprobs(actor,ids,attn,mask)
                mismatches.append((anchor-sampled).cpu())
                off=0
                for b,_,_ in chunk:
                    b.training_old_logprobs=anchor[off:off+b.ntokens].cpu().tolist();off+=b.ntokens
        actor.train()
    if mismatches:
        summary=distribution(torch.cat(mismatches))
        metrics.update({'rollout_logprob_delta_'+k:v for k,v in summary.items() if k!='finite'})
    gradient_records=[]
    for chunks in stages:
        optimizer.zero_grad(set_to_none=True)
        low=cfg['dapo_clip_low'] if method=='dapo' else cfg['clip_low']
        total_loss=0.
        for ci,chunk in enumerate(chunks):
            ids,attn,mask,adv,weight,old=pack(chunk,tokenizer.pad_token_id)
            if cfg.get('kl_estimator','sampled')=='categorical':
                current,kl=selected_exact_kl(actor,ref,ids,attn,mask)
            else:
                with torch.no_grad():ref_lp=selected_logprobs(ref,ids,attn,mask)
                current=selected_logprobs(actor,ids,attn,mask,gradient=True)
                logratio=ref_lp-current
                kl=torch.expm1(logratio)-logratio
            ratio=torch.exp(current-old)
            pg=-torch.minimum(ratio*adv,torch.clamp(ratio,1-low,1+cfg['clip_high'])*adv)
            correction=torch.ones_like(pg)
            if cfg.get('rollout_is_max'):
                sampled=torch.tensor([v for b,_,_ in chunk for v,m in zip(b.old_logprobs,b.mask) if m],dtype=torch.float32,device='cuda')
                correction=torch.exp((old-sampled).clamp(-20,20)).clamp(max=cfg['rollout_is_max']).detach()
                metrics['rollout_is_clipped_weight']+=((old-sampled>np.log(cfg['rollout_is_max'])).float()*weight).sum().item()
            pg_loss=(pg*correction*weight).sum()
            kl_loss=cfg['kl_coefficient']*(kl*weight).sum()
            loss=pg_loss+kl_loss
            if not torch.isfinite(loss):raise FloatingPointError('Nonfinite loss')
            if ci==0 and cfg.get('diagnostic_gradient_split',False):
                probe=actor.model.layers[-1].mlp.down_proj.weight
                pg_g=torch.autograd.grad(pg_loss,probe,retain_graph=True)[0]
                kl_g=torch.autograd.grad(kl_loss,probe,retain_graph=True)[0]
                gradient_records.append(dict(probe_pg_norm=pg_g.norm().item(),probe_weighted_kl_norm=kl_g.norm().item()))
                del pg_g,kl_g
            loss.backward()
            total_loss+=loss.detach().item()
            metrics['pg_loss']+=pg_loss.detach().item()
            metrics['kl']+=(kl.detach()*weight).sum().item()
            metrics['max_token_kl']=max(metrics['max_token_kl'],kl.detach().max().item())
            metrics['max_abs_log_ratio']=max(metrics['max_abs_log_ratio'],(current-old).detach().abs().max().item())
            metrics['clipfrac']+=(((ratio.detach()<1-low)|(ratio.detach()>1+cfg['clip_high'])).float()*weight).sum().item()
            metrics['sampled_surprisal']+=((-current.detach())*weight).sum().item()
            metrics['weighted_response_tokens']+=len(old)
            del ids,attn,mask,adv,weight,old,current,ratio,pg,kl,loss,pg_loss,kl_loss,correction
        norm=torch.nn.utils.clip_grad_norm_(actor.parameters(),cfg['max_grad_norm'])
        if not torch.isfinite(norm):raise FloatingPointError('Nonfinite gradient norm')
        probe=actor.model.layers[-1].mlp.down_proj.weight
        before=probe.detach().clone()
        optimizer.step()
        delta=(probe.detach()-before).abs().max().item()
        metrics['max_parameter_change']=max(metrics['max_parameter_change'],delta)
        metrics['loss']+=total_loss
        metrics['grad_norm']+=float(norm)
        metrics['clipped_optimizer_updates']+=int(norm>cfg['max_grad_norm'])
        mb_count+=1
    optimizer.zero_grad(set_to_none=True)
    actor.eval()
    for key in ['loss','pg_loss','grad_norm','kl','clipfrac','sampled_surprisal','rollout_is_clipped_weight']:
        metrics[key]/=mb_count
    result=dict(metrics)
    result['optimizer_updates']=mb_count
    result['probe_gradient_split']=gradient_records
    result['kl_estimator']=cfg.get('kl_estimator','sampled')
    return result


def save_checkpoint(path,actor,optimizer,step,rng,cfg,elapsed):
    path=Path(path)
    path.parent.mkdir(parents=True,exist_ok=True)
    # torch.save handles CUDA tensors; restore maps to CPU before copying to model.
    state={'step':step,'model':actor.state_dict(),'optimizer':optimizer.state_dict(),
           'rng':rng.bit_generator.state,'torch_rng':torch.get_rng_state(),
           'cuda_rng':torch.cuda.get_rng_state_all(),'python_rng':random.getstate(),
           'config':cfg,'elapsed_seconds':elapsed}
    atomic_checkpoint(path,state)


def evaluate(engine,tokenizer,cfg,tasks,out_path,seed,split):
    inference_cache_boundary(engine,'before_evaluation',reset=False)
    rng=np.random.default_rng(seed)
    start=time.monotonic()
    branches,stats,_=collect(tasks,engine,tokenizer,cfg,'grpo',rng,evaluation=True)
    inference_cache_boundary(engine,'after_evaluation')
    metrics=evaluation_metrics(branches,tasks)
    metrics.update(split=split,seconds=time.monotonic()-start,generated_tokens=stats['generated_tokens'],eos_token_id=tokenizer.eos_token_id)
    atomic_json(out_path,metrics)
    records=Path(out_path).with_suffix('.jsonl')
    with records.open('w') as f:
        for b in branches:
            f.write(json.dumps(dict(task=tasks[b.task_id],lineage=b.lineage,reward=b.reward,
                         terminal=b.terminal,turns=b.turns,response_tokens=b.ntokens,trace=b.trace,
                         ids=b.ids,mask=b.mask,old_logprobs=b.old_logprobs))+'\n')
    return metrics


def write_rollouts(path,branches,tasks,mode='wt',attempt=None):
    with gzip.open(path,mode) as f:
        for b in branches:
            f.write(json.dumps(dict(task=tasks[b.task_id],lineage=b.lineage,reward=b.reward,
                terminal=b.terminal,ids=b.ids,mask=b.mask,old_logprobs=b.old_logprobs,
                trace=b.trace,resample_attempt=attempt,task_id=b.task_id,
                training_reward=getattr(b,'training_reward',b.reward)))+'\n')


def audit_saved_validation(path,tasks,cfg):
    """Independently replay an earlier saved evaluation before resuming."""
    from scripts.verify_final import audit_evaluation
    return audit_evaluation(path,tasks,cfg,'validation')


def redo_validation_after_resume(engine,tokenizer,cfg,tasks,out,step):
    """Keep retry cost evidence even if evaluation or its output write fails."""
    import uuid
    attempt=out/'recovery_evaluations'/(uuid.uuid4().hex+'.json')
    evidence=dict(status='started',validation_step=step,started_unix=time.time(),
        measured_wall_seconds=None,recorded_metric_seconds=None,generated_tokens=None,
        cost_scope='This retry evaluation call, including result writes; model loading and earlier discarded attempts are separate. A killed started attempt has unknown cost.')
    atomic_json(attempt,evidence)
    started=time.monotonic()
    try:
        metrics=evaluate(engine,tokenizer,cfg,tasks,out/f'validation_{step:04d}.json',100042,'validation')
    except BaseException as exc:
        evidence.update(status='failed',measured_wall_seconds=time.monotonic()-started,error=repr(exc))
        atomic_json(attempt,evidence)
        raise
    evidence.update(status='complete',measured_wall_seconds=time.monotonic()-started,
        recorded_metric_seconds=metrics.get('seconds'),generated_tokens=metrics.get('generated_tokens'))
    atomic_json(attempt,evidence)
    return metrics


def main():
    parser=argparse.ArgumentParser()
    parser.add_argument('--method',required=True,choices=['grpo','dapo','arpo','tree_random','patr'])
    parser.add_argument('--config',required=True)
    parser.add_argument('--run-root',default='runs/short_validation')
    parser.add_argument('--smoke',action='store_true')
    parser.add_argument('--resume',action='store_true')
    parser.add_argument('--stop-after-step',type=int)
    args=parser.parse_args()
    cfg=json.loads((ROOT/args.config).read_text())
    override_path=ROOT/'artifacts/runtime_override.json'
    if override_path.exists():
        override=json.loads(override_path.read_text())
        assert set(override)<=OVERRIDE_KEYS,'Only engineering batch/memory/execution overrides are allowed'
        cfg.update(override)
    if args.smoke:
        cfg.update(steps=1,train_batch_size=4,mini_batch_tasks=4,eval_every=1,save_every=1,
                   max_response_tokens=1024)
    if cfg.get('run_mode')!='short_validation' or not 1<=cfg['steps']<=40:
        raise ValueError('This repair checkout only permits explicit short_validation runs of 1..40 steps')
    seed=cfg['seed']
    cfg['active_method']=args.method
    out=ROOT/args.run_root/f'{args.method}_seed{seed}'
    data=make_datasets(cfg)
    data_path=ROOT/cfg.get('dataset_path','artifacts/data.json')
    fingerprint=build_fingerprint(dict(cfg,method=args.method),data_path,root=ROOT)
    validation=data['validation'][:cfg.get('validation_tasks',100)]
    preparation=prepare_run(out,args.resume,fingerprint,dict(cfg,method=args.method),root=ROOT,
        evaluation_validator=lambda path:audit_saved_validation(path,validation,cfg))
    atomic_json(out/'status.json',dict(status='initializing',method=args.method,step=preparation['start_step'],total_steps=cfg['steps'],pid=os.getpid(),run_mode='short_validation'))
    actor,ref,engine,tokenizer=build_models(cfg,seed)
    vocabulary=configure_vocabulary(actor,ref,tokenizer,cfg,args.method)
    atomic_json(out/'vocabulary.json',vocabulary)
    optimizer=torch.optim.AdamW(actor.parameters(),lr=cfg['learning_rate'],betas=(.9,.999),eps=1e-8,weight_decay=0.)
    rng=np.random.default_rng(seed)
    start_step=0;previous_elapsed=0.
    cp=out/'checkpoint.pt'
    if args.resume and cp.exists():
        state=torch.load(cp,map_location='cpu',weights_only=False)
        assert {k:v for k,v in state['config'].items() if k not in RUNTIME_KEYS}=={k:v for k,v in cfg.items() if k not in RUNTIME_KEYS},'Cannot resume after an algorithm configuration change'
        actor.load_state_dict(state['model']);optimizer.load_state_dict(state['optimizer'])
        start_step=state['step'];rng.bit_generator.state=state['rng']
        torch.set_rng_state(state['torch_rng']);torch.cuda.set_rng_state_all(state['cuda_rng']);random.setstate(state['python_rng'])
        previous_elapsed=state.get('elapsed_seconds',0.)
        del state
        sync_weights(actor,engine)
        print(json.dumps({'RESUMED':start_step,'checkpoint':str(cp)}),flush=True)
    signal.signal(signal.SIGTERM,request_stop)
    signal.signal(signal.SIGINT,request_stop)
    os.environ['PATR_PROGRESS_PATH']=str(out/'progress.json')
    # Freeze test use to final evaluation. Training curves use a separate validation set.
    pending_validation=preparation.get('pending_validation_step')
    if pending_validation is not None:
        deadline_file=ROOT/'artifacts/deadline.json'
        deadline=json.loads(deadline_file.read_text())['unix'] if deadline_file.exists() else float('inf')
        if STOP or time.time()>=deadline:
            atomic_json(out/'status.json',dict(status='paused_checkpointed',step=start_step,total_steps=cfg['steps'],method=args.method,
                validation_pending=pending_validation,final_evaluation_pending=start_step==cfg['steps']))
            return
        repaired=redo_validation_after_resume(engine,tokenizer,cfg,validation,out,pending_validation)
        print(json.dumps({'resumed_validation_step':pending_validation,**repaired}),flush=True)
    elif start_step==0:
        # A fresh run commits its initialized actor/optimizer/RNG once,
        # before a potentially long initial validation. CP0 resumes use
        # the pending-validation branch above and never recommit CP0.
        initial_checkpoint_started=time.monotonic()
        save_checkpoint(cp,actor,optimizer,0,rng,cfg,previous_elapsed)
        atomic_json(out/'initial_checkpoint.json',dict(step=0,checkpoint_bytes=cp.stat().st_size,
            checkpoint_write_seconds=time.monotonic()-initial_checkpoint_started,
            scope='Initial CP0 serialization/fsync/atomic commit only; excludes model/CUDA initialization and later evaluation. This live checkpoint is replaced by later committed steps.'))
        deadline_file=ROOT/'artifacts/deadline.json'
        deadline=json.loads(deadline_file.read_text())['unix'] if deadline_file.exists() else float('inf')
        if STOP or time.time()>=deadline:
            atomic_json(out/'status.json',dict(status='paused_checkpointed',step=0,total_steps=cfg['steps'],method=args.method,
                validation_pending=0,final_evaluation_pending=False))
            return
        base=evaluate(engine,tokenizer,cfg,validation,out/'validation_0000.json',100042,'validation')
        print(json.dumps({'initial_validation':base}),flush=True)
    start=time.monotonic()
    step=start_step
    for step in range(start_step+1,cfg['steps']+1):
        deadline_file=ROOT/'artifacts/deadline.json'
        deadline=json.loads(deadline_file.read_text())['unix'] if deadline_file.exists() else float('inf')
        if STOP or time.time()>=deadline:
            step-=1
            break
        atomic_json(out/'status.json',dict(status='training',method=args.method,step=step-1,
                      current_step=step,total_steps=cfg['steps'],pid=os.getpid()))
        t0=time.monotonic()
        # Keep the task stream independent of method-specific branching RNG consumption.
        task_rng=np.random.default_rng(np.random.SeedSequence([seed,step,1701]))
        task_indices=task_rng.choice(len(data['train']),cfg['train_batch_size'],replace=False).tolist()
        tasks=[data['train'][i] for i in task_indices]
        branches,stats,events=collect(tasks,engine,tokenizer,cfg,args.method,rng)
        if args.method=='dapo':
            # Rejection sampling of zero-variance prompt groups; charge all rejected rollout work.
            accepted=[];accepted_tasks=[];accepted_uids=set();attempts=1
            candidate_indices=set(task_indices)
            while True:
                write_rollouts(out/f'dapo_candidates_{step:04d}.jsonl.gz',branches,tasks,
                               'wt' if attempts==1 else 'at',attempts)
                groups={ti:[b for b in branches if b.task_id==ti] for ti in range(len(tasks))}
                for ti,task in enumerate(tasks):
                    bs=groups[ti]
                    if len({getattr(b,'training_reward',None) if getattr(b,'training_reward',None) is not None else b.reward for b in bs})>1 and task['uid'] not in accepted_uids and len(accepted_tasks)<cfg['train_batch_size']:
                        for b in bs:b.task_id=len(accepted_tasks)
                        accepted.extend(bs);accepted_tasks.append(task);accepted_uids.add(task['uid'])
                if len(accepted_tasks)==cfg['train_batch_size']:
                    break
                if attempts>=cfg['dapo_max_resample_batches']:
                    raise RuntimeError('DAPO dynamic sampling could not fill batch; no silent fallback')
                remaining=[i for i in range(len(data['train'])) if i not in candidate_indices]
                if len(remaining)<cfg['train_batch_size']:raise RuntimeError('DAPO exhausted unique candidate tasks')
                next_indices=task_rng.choice(remaining,cfg['train_batch_size'],replace=False).tolist()
                candidate_indices.update(next_indices)
                tasks=[data['train'][i] for i in next_indices]
                branches,more,ev=collect(tasks,engine,tokenizer,cfg,'dapo',rng)
                n0,n1=stats['episodes'],more['episodes']
                for k in ['success_rate','mean_response_tokens','mean_turns','mean_training_reward','first_turn_invalid_fraction']:
                    stats[k]=(stats[k]*n0+more[k]*n1)/(n0+n1)
                stats['terminal_counts']=dict(Counter(stats['terminal_counts'])+Counter(more['terminal_counts']))
                for k in ['generated_tokens','input_tokens','episodes','executed_actions','invalid_actions']:stats[k]+=more[k]
                stats['invalid_action_fraction']=stats['invalid_actions']/stats['executed_actions'] if stats['executed_actions'] else None
                events.extend(ev);attempts+=1
            branches=accepted;tasks=accepted_tasks
            stats['resample_batches']=attempts
            stats['accepted_success_rate']=float(np.mean([b.reward>0 for b in branches]))
        outcome_groups=[[b.reward for b in branches if b.task_id==ti] for ti in range(len(tasks))]
        learning_groups=[[getattr(b,'training_reward',None) if getattr(b,'training_reward',None) is not None else b.reward for b in branches if b.task_id==ti] for ti in range(len(tasks))]
        stats['outcome_zero_variance_group_fraction']=sum(len(set(g))==1 for g in outcome_groups)/len(tasks)
        stats['learning_zero_variance_group_fraction']=sum(len(set(g))==1 for g in learning_groups)/len(tasks)
        stats['accepted_distinct_task_uids']=len({task['uid'] for task in tasks})
        sampled_time=time.monotonic()-t0
        atomic_json(out/'progress.json',dict(phase='optimization',method=args.method,step=step,
                      trajectories=len(branches),updated_unix=time.time()))
        optimizer_rng=np.random.default_rng(np.random.SeedSequence([seed,step,1702]))
        changes=update(actor,ref,optimizer,branches,cfg,args.method,tokenizer,optimizer_rng,len(tasks))
        sync_weights(actor,engine)
        elapsed=previous_elapsed+time.monotonic()-start
        record=dict(step=step,method=args.method,seed=seed,rollout_seconds=sampled_time,
                    resumed_from_step=start_step,
                    step_seconds=time.monotonic()-t0,elapsed_seconds=elapsed,
                    peak_allocated_gib=torch.cuda.max_memory_allocated()/2**30,
                    time_utc=time.strftime('%Y-%m-%dT%H:%M:%SZ',time.gmtime()),**stats,**changes)
        append_json(out/'train_metrics.jsonl',record)
        append_json(out/'branch_events.jsonl',dict(step=step,events=events))
        # Preserve complete raw sampled trajectories for auditing, including failures/pruning.
        write_rollouts(out/f'rollouts_{step:04d}.jsonl.gz',branches,tasks)
        print(json.dumps(record),flush=True)
        atomic_json(out/'status.json',dict(status='training',method=args.method,step=step,total_steps=cfg['steps'],pid=os.getpid(),last=record))
        # Protect the first completed update before the regular save interval.
        if step==1 or step%cfg['save_every']==0 or step==cfg['steps']:
            save_checkpoint(cp,actor,optimizer,step,rng,cfg,elapsed)
        if step%cfg['eval_every']==0 or step==cfg['steps']:
            deadline_file=ROOT/'artifacts/deadline.json'
            deadline=json.loads(deadline_file.read_text())['unix'] if deadline_file.exists() else float('inf')
            if STOP or time.time()>=deadline:
                if not (step==1 or step%cfg['save_every']==0 or step==cfg['steps']):
                    save_checkpoint(cp,actor,optimizer,step,rng,cfg,previous_elapsed+time.monotonic()-start)
                atomic_json(out/'status.json',dict(status='paused_checkpointed',step=step,total_steps=cfg['steps'],method=args.method,
                    validation_pending=step,final_evaluation_pending=step==cfg['steps']))
                return
            m=evaluate(engine,tokenizer,cfg,validation,out/f'validation_{step:04d}.json',100042,'validation')
            print(json.dumps({'validation_step':step,**m}),flush=True)
        if args.stop_after_step and step>=args.stop_after_step:
            break
    # A completed short checkpoint may still have final validation pending.
    # Respect the run window before starting that separate evaluation.
    deadline_file=ROOT/'artifacts/deadline.json'
    deadline=json.loads(deadline_file.read_text())['unix'] if deadline_file.exists() else float('inf')
    if step<cfg['steps'] or STOP or time.time()>=deadline:
        save_checkpoint(cp,actor,optimizer,step,rng,cfg,previous_elapsed+time.monotonic()-start)
        atomic_json(out/'status.json',dict(status='paused_checkpointed',step=step,total_steps=cfg['steps'],method=args.method,
                    final_evaluation_pending=step==cfg['steps']))
        return
    # Short validation never evaluates the reserved test/holdout or launches a formal queue.
    final=evaluate(engine,tokenizer,cfg,validation,out/'final_validation.json',100042,'short_validation')
    atomic_json(out/'status.json',dict(status='complete',run_mode='short_validation',step=step,total_steps=cfg['steps'],method=args.method,final=final))
    print(json.dumps({'FINAL':final}),flush=True)


if __name__=='__main__':
    main()
