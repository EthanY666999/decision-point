class Decoder:
    def decode(self,ids,**kwargs):
        assert kwargs==dict(skip_special_tokens=True,clean_up_tokenization_spaces=False)
        return '```Right```' if ids[0]==10 else '```Left```'

    yield fixture
    fixture.doCleanups()


def test_verified_decoder_matches_saved_response_and_discloses_helper_scope(evaluation):
    f=evaluation
    result=verify.audit_evaluation(f.path,[f.task],f.cfg,'validation',tokenizer=Decoder())
    assert result['response_text_decode_verified'] is True
    assert result['response_text_decode_modes']=={'standalone':4}
    without_decoder=verify.audit_evaluation(f.path,[f.task],f.cfg,'validation')
    assert without_decoder['response_text_decode_verified'] is False


def test_same_length_ids_with_different_decoded_action_are_rejected(evaluation,monkeypatch):
    f=evaluation
    f.records[0]['ids'][1]=12
    f.records[0]['trace'][0]['response_token_ids'][0]=12
    f.write()
    # Length/mask/trace IDs agree, and saved text still parses Right and succeeds.
    assert verify.audit_evaluation(f.path,[f.task],f.cfg,'validation')['status']=='verified'
    before={path:path.read_bytes() for path in (f.path,f.path.with_suffix('.jsonl'))}
    prefixes=[]
    def replay(tokenizer,prompt_ids,response_ids):
        prefixes.append(prompt_ids)
        return tokenizer.decode(response_ids,skip_special_tokens=True,clean_up_tokenization_spaces=False)
    monkeypatch.setattr(verify,'incremental_response_text',replay)
    with pytest.raises(ValueError,match='decode to different saved text'):
        verify.audit_evaluation(f.path,[f.task],f.cfg,'validation',tokenizer=Decoder())
    assert prefixes==[f.records[0]['ids'][:1]]
    assert before=={path:path.read_bytes() for path in before}


def test_incremental_mode_reports_exact_prefix_and_never_strips_saved_text(evaluation,monkeypatch):
    f=evaluation
    class TailDecoder(Decoder):
        def decode(self,ids,**kwargs):
            return super().decode(ids,**kwargs)+'\ufffd'
    calls=[]
    def replay(tokenizer,prompt_ids,response_ids):
        calls.append((prompt_ids,response_ids))
        return '```Right```'
    monkeypatch.setattr(verify,'incremental_response_text',replay)
    result=verify.audit_evaluation(f.path,[f.task],f.cfg,'validation',tokenizer=TailDecoder())
    assert result['response_text_decode_modes']=={'incremental':4}
    assert calls==[(record['ids'][:1],record['trace'][0]['response_token_ids']) for record in f.records]
    # Exact replay, not a suffix-normalization exception: an extra space fails.
    with pytest.raises(ValueError,match='both standalone and exact vLLM'):
        verify.verify_response_text(TailDecoder(),[1],[10],'```Right``` ')

def test_actual_saved_utf8_tail_requires_exact_vllm_incremental_replay(real_unicode_tail):
    evidence,tokenizer=real_unicode_tail
    prefix=evidence['prompt_token_ids'];ids=evidence['response_token_ids']
    saved=evidence['expected_incremental_text']
    standalone=tokenizer.decode(ids,skip_special_tokens=True,clean_up_tokenization_spaces=False)
    assert standalone==evidence['standalone_text'] and standalone!=saved
    assert verify.incremental_response_text(tokenizer,prefix,ids)==saved
    assert verify.verify_response_text(tokenizer,prefix,ids,saved)=='incremental'


def test_real_tail_forged_standalone_text_cannot_bypass_runtime_decoder(real_unicode_tail):
    evidence,tokenizer=real_unicode_tail
    forged=evidence['standalone_text']
    assert '\ufffd' in forged
    assert tokenizer.decode(evidence['response_token_ids'],skip_special_tokens=True,
        clean_up_tokenization_spaces=False)==forged
    with pytest.raises(ValueError,match='both standalone and exact vLLM'):
        verify.verify_response_text(tokenizer,evidence['prompt_token_ids'],evidence['response_token_ids'],forged)


@pytest.mark.parametrize('replacement',['Right','Left'])
def test_real_tail_case_forged_same_length_text_is_not_accepted(real_unicode_tail,replacement):
    evidence,tokenizer=real_unicode_tail
    saved=evidence['expected_incremental_text']
    forged=replacement+saved[len(replacement):]
    assert len(forged)==len(saved) and forged!=saved
    with pytest.raises(ValueError,match='both standalone and exact vLLM'):
        verify.verify_response_text(tokenizer,evidence['prompt_token_ids'],evidence['response_token_ids'],forged)

@pytest.mark.parametrize('missing',[None,(0,'json'),(0,'jsonl'),(10,'json'),(10,'jsonl'),(20,'json'),(20,'jsonl')])
def test_formal_cli_decodes_complete_schedule_and_rejects_missing_primary(tmp_path,monkeypatch,evaluation,missing):
    f=evaluation;root=tmp_path;run=root/'runs/smoke/grpo_seed42';run.mkdir(parents=True)
    cfg=dict(f.cfg,methods=['grpo'],seed=42,model='fixture',steps=20,eval_every=10,train_batch_size=4,mini_batch_tasks=4,run_mode='short_validation',validation_tasks=1)
    config_path=root/'config.json';config_path.write_text(json.dumps(cfg))
    data_path=root/'data.json';data_path.write_text(json.dumps({'validation':[f.task],'test':[]}))
    run_cfg=dict(cfg,method='grpo')
    (run/'config.json').write_text(json.dumps(run_cfg))
    (run/'fingerprint.json').write_text(json.dumps({'sha256':'fixture-fingerprint','data_sha256':hashlib.sha256(data_path.read_bytes()).hexdigest()}))
    (run/'status.json').write_text('{"status":"complete","step":20}')
    (run/'train_metrics.jsonl').write_text(''.join(json.dumps(dict(step=step,loss=.1,kl=.01,grad_norm=.2))+'\n' for step in range(1,21)))
    for name,split in [('validation_0000','validation'),('validation_0010','validation'),('validation_0020','validation'),('final_validation','short_validation')]:
        (run/(name+'.json')).write_text(json.dumps(dict(f.metrics,split=split)))
        (run/(name+'.jsonl')).write_bytes(f.path.with_suffix('.jsonl').read_bytes())
    torch.save(dict(step=20,integrity_fingerprint='fixture-fingerprint',config=cfg,
                    optimizer={'state':{0:{'step':torch.tensor(20.)}}}),run/'checkpoint.pt')
    monkeypatch.setattr(verify,'ROOT',root)
    calls=[]
    def loader(actual_cfg,fp):
        calls.append(actual_cfg['method']);return Decoder(),{'fixture_verified_assets':True}
    monkeypatch.setattr(verify,'load_verified_tokenizer',loader)
    output=root/'audit.json'
    monkeypatch.setattr(sys,'argv',['verify_final.py','--config',str(config_path),'--run-root',str(run.parent),
        '--data',str(data_path),'--include-validation','--require-complete','--output',str(output)])
    if missing:
        step,suffix=missing
        (run/f'validation_{step:04d}.{suffix}').unlink()
        before={p.name:p.read_bytes() for p in run.iterdir() if p.is_file()}
        with pytest.raises(ValueError,match='Missing required scheduled validation'):
            verify.main()
        assert before=={p.name:p.read_bytes() for p in run.iterdir() if p.is_file()} and not output.exists()
        return
    verify.main()
    report=json.loads(output.read_text())
    assert report['all_selected_verified'] and calls==['grpo']
    assert report['formal'][0]['response_text_decode_verified'] and report['validation'][0]['response_text_decode_verified']
    assert report['tokenizer_identity']=={'grpo':{'fixture_verified_assets':True}}
    assert len(report['validation'])==3 and report['validation_schedule']['grpo']['required_now']==[0,10,20]


@pytest.mark.parametrize('require_complete',[False,True])
@pytest.mark.parametrize('step',[0,10,20])
def test_cli_paused_pending_boundary_requires_only_earlier_schedule_and_never_reports_complete(tmp_path,monkeypatch,evaluation,require_complete,step):
    f=evaluation;run=tmp_path/'runs/paused/grpo_seed42';run.mkdir(parents=True)
    cfg=dict(f.cfg,methods=['grpo'],seed=42,model='fixture',steps=20,eval_every=10,
             train_batch_size=4,mini_batch_tasks=4,run_mode='short_validation',validation_tasks=1)
    config=tmp_path/'config.json';config.write_text(json.dumps(cfg))
    data=tmp_path/'data.json';data.write_text(json.dumps({'validation':[f.task],'test':[]}))
    (run/'config.json').write_text(json.dumps(dict(cfg,method='grpo')))
    (run/'fingerprint.json').write_text(json.dumps({'sha256':'fixture','data_sha256':hashlib.sha256(data.read_bytes()).hexdigest()}))
    (run/'status.json').write_text(json.dumps(dict(status='paused_checkpointed',step=step,validation_pending=step)))
    (run/'train_metrics.jsonl').write_text(''.join(json.dumps(dict(step=value,loss=.1,kl=.01,grad_norm=.2))+'\n' for value in range(1,step+1)))
    for prior in [value for value in (0,10,20) if value<step]:
        (run/f'validation_{prior:04d}.json').write_text(json.dumps(f.metrics))
        (run/f'validation_{prior:04d}.jsonl').write_bytes(f.path.with_suffix('.jsonl').read_bytes())
    # A pending boundary can contain a failed partial attempt; disclose rather
    # than count it among verified earlier evaluations.
    (run/f'validation_{step:04d}.json').write_bytes(b'{"partial":')
    monkeypatch.setattr(verify,'ROOT',tmp_path)
    monkeypatch.setattr(verify,'load_verified_tokenizer',lambda *args:(Decoder(),{'fixture':True}))
    output=tmp_path/'audit.json'
    args=['verify_final.py','--config',str(config),'--run-root',str(run.parent),'--data',str(data),
          '--include-validation','--output',str(output)]
    if require_complete:args.append('--require-complete')
    monkeypatch.setattr(sys,'argv',args)
    if require_complete:
        with pytest.raises(SystemExit,match='remain incomplete'):
            verify.main()
    else:verify.main()
    result=json.loads(output.read_text())
    expected_prior=[value for value in (0,10,20) if value<step]
    assert result['all_selected_verified'] is False and len(result['validation'])==len(expected_prior)
    assert result['formal'][0]['status']=='pending'
    scope=result['validation_schedule']['grpo']
    assert scope['required_now']==expected_prior and scope['deferred_steps']==[value for value in (0,10,20) if value>=step]
    assert scope['existing_files_outside_verified_schedule']==[f'validation_{step:04d}.json']
